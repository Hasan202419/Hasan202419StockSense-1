class AIStockScreener:
    def __init__(self):
        self.sentiment_analyzer = pipeline("text-classification", 
                                          model="mrm8488/distilroberta-finetuned-financial-news-sentiment")
        self.model = self._train_model()
        self.last_scan = {}
        
    def _train_model(self):
        # In production, use historical data with features: [volume_change, news_sentiment, options_volume, rsi, etc.]
        X = np.random.rand(100, 5)
        y = np.random.randint(2, size=100)
        return GradientBoostingClassifier().fit(X, y)
    
    def scan_market(self):
        """Scan all US stocks for buy signals"""
        # Get all tickers (in reality, use comprehensive list)
        all_tickers = self._get_all_tickers()
        signals = []
        
        for ticker in all_tickers:
            try:
                # Get real-time data
                stock = yf.Ticker(ticker)
                hist = stock.history(period="5d", interval="15m")
                
                if hist.empty:
                    continue
                    
                # 1. Volume spike detection
                current_volume = hist['Volume'][-1]
                avg_volume = np.mean(hist['Volume'][:-1])
                volume_ratio = current_volume / avg_volume
                
                # 2. News analysis
                news_sentiment = self._analyze_news(stock)
                
                # 3. Options activity
                options_activity = self._check_options_chain(ticker)
                
                # 4. Generate signal
                features = np.array([[volume_ratio, news_sentiment, options_activity, 
                                    self._calculate_rsi(hist), self._get_short_interest(ticker)]])
                probability = self.model.predict_proba(features)[0][1]
                
                if probability > 0.75:
                    signals.append({
                        'ticker': ticker,
                        'probability': probability,
                        'price': hist['Close'][-1],
                        'volume_ratio': volume_ratio,
                        'news_sentiment': news_sentiment,
                        'options_activity': options_activity,
                        'timestamp': datetime.now()
                    })
                    
            except Exception as e:
                print(f"Error scanning {ticker}: {str(e)}")
                
        self.last_scan = {
            'timestamp': datetime.now(),
            'signals': signals
        }
        return signals
    
    def _get_all_tickers(self):
        """Get all US-listed tickers (simplified for demo)"""
        return ['AAPL', 'TSLA', 'AMZN', 'MSFT', 'GOOGL', 'SNDL', 'CEI', 'BNGO', 'ATER', 'MMAT', 'BBIG']
    
    def _analyze_news(self, stock):
        """Analyze news sentiment for a stock"""
        try:
            news = stock.news
            if not news:
                return 0.5
                
            sentiments = []
            for item in news[:3]:
                result = self.sentiment_analyzer(item['title'])
                score = result[0]['score']
                sentiment = score if result[0]['label'] == 'positive' else 1 - score
                sentiments.append(sentiment)
                
            return np.mean(sentiments)
        except:
            return 0.5
    
    def _check_options_chain(self, ticker):
        """Check for unusual options activity"""
        try:
            stock = yf.Ticker(ticker)
            options = stock.option_chain()
            calls = options.calls
            puts = options.puts
            
            if calls.empty or puts.empty:
                return 0
                
            call_volume = calls['volume'].sum()
            put_volume = puts['volume'].sum()
            put_call_ratio = put_volume / call_volume if call_volume > 0 else 0
            
            # Higher call volume = bullish signal
            return 1 - min(put_call_ratio, 1)
        except:
            return 0
    
    def _calculate_rsi(self, data, window=14):
        """Calculate Relative Strength Index"""
        delta = data['Close'].diff()
        gain = (delta.where(delta > 0, 0)).rolling(window=window).mean()
        loss = (-delta.where(delta < 0, 0)).rolling(window=window).mean()
        
        rs = gain / loss
        return 100 - (100 / (1 + rs)) if loss.any() else 50
    
    def _get_short_interest(self, ticker):
        """Get short interest ratio (simulated)"""
        return np.random.rand()  # In reality, use API data
